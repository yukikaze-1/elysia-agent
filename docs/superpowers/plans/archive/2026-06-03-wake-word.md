```python
"""Wake word detector using openWakeWord — streaming, on-device detection.

openWakeWord requires 80ms frames of 16kHz mono Int16 PCM (1280 samples = 2560 bytes).
AudioBuffer sends 256ms chunks — WakeWordDetector buffers and slices internally.
"""

import logging
import time
from typing import Callable

import numpy as np

logger = logging.getLogger(__name__)

# openWakeWord frame: 80ms at 16kHz mono Int16
_WW_FRAME_MS = 80
_WW_FRAME_SAMPLES = int(16000 * _WW_FRAME_MS / 1000)  # 1280
_WW_FRAME_BYTES = _WW_FRAME_SAMPLES * 2                 # 2560

# Detection threshold (openWakeWord returns 0-1 score)
_SCORE_THRESHOLD = 0.5
# Debounce: ignore hits within this many seconds of a previous hit
_DEBOUNCE_SECONDS = 1.0


class WakeWordDetector:
    """Streaming wake word detector wrapping openWakeWord.

    Buffers incoming 256ms PCM chunks, slices into 80ms frames, and feeds each
    frame to one or more openWakeWord models. Fires on_wake callback on detection.
    """

    def __init__(
        self,
        wake_words: list[str],
        model_paths: dict[str, str] | None = None,
        on_wake: Callable[[], object] | None = None,
    ):
        self._wake_words = wake_words
        self._model_paths = model_paths or {}
        self._on_wake = on_wake
        self._buffer: bytearray = bytearray()
        self._models: list[tuple[str, object]] = []  # [(word, model), ...]
        self._last_wake: float = 0.0

        self._load_models()

    def _load_models(self) -> None:
        try:
            import openwakeword
        except ImportError:
            logger.warning(
                "openwakeword not installed. Install with: pip install openwakeword numpy. "
                "Wake word detection disabled."
            )
            return

        for word in self._wake_words:
            try:
                model_path = self._model_paths.get(word)
                if model_path:
                    model = openwakeword.Model(wakeword_models=[model_path])
                else:
                    model = openwakeword.Model(wakeword_models=[word])
                self._models.append((word, model))
                logger.info("Wake word model loaded: %s", word)
            except Exception:
                logger.warning(
                    "Failed to load wake word model for '%s'. "
                    "Provide a .tflite path via model_paths for custom words.", word
                )

    @property
    def enabled(self) -> bool:
        return len(self._models) > 0

    def feed(self, chunk: bytes) -> None:
        if not self._models:
            return

        self._buffer.extend(chunk)

        while len(self._buffer) >= _WW_FRAME_BYTES:
            frame = bytes(self._buffer[:_WW_FRAME_BYTES])
            self._buffer = self._buffer[_WW_FRAME_BYTES:]

            samples = np.frombuffer(frame, dtype=np.int16)
            for word, model in self._models:
                try:
                    prediction = model.predict(samples)
                    score = prediction.get(word, 0.0)
                except Exception:
                    continue

                if score > _SCORE_THRESHOLD:
                    now = time.monotonic()
                    if now - self._last_wake >= _DEBOUNCE_SECONDS:
                        self._last_wake = now
                        logger.info("Wake word detected: %s (score=%.3f)", word, score)
                        if self._on_wake:
                            self._on_wake()

    def reset(self) -> None:
        self._buffer.clear()

    def close(self) -> None:
        self._models.clear()
        self._buffer.clear()
```

        """Feeding exactly one 80ms frame consumes it entirely."""
        detector = WakeWordDetector(wake_words=["test"])
        detector.feed(_silence_bytes(1280))  # 1280 samples = 2560 bytes = 80ms
        assert len(detector._buffer) == 0

        """256ms chunk splits into three 80ms frames (2560*3=7680, 8192-7680=512 remainder)."""

        detector.feed(_loud_bytes(4096))  # 256ms chunk
        # 8192 bytes → 3 × 2560 = 7680 consumed, 512 remaining
        assert len(detector._buffer) == 512

    def test_partial_buffer_accumulates(self):
        """40ms chunk (< 80ms) stays in buffer."""
        detector = WakeWordDetector(wake_words=["test"])

        assert len(detector._buffer) == 1280  # 640 samples × 2 bytes

    def test_reset_clears_buffer(self):
        detector = WakeWordDetector(wake_words=["test"])
        detector.feed(_silence_bytes(1280))
        assert len(detector._buffer) > 0
        detector.reset()
        assert len(detector._buffer) == 0

    async def feed(self, chunk: bytes) -> None:
        """Feed a raw Int16 PCM chunk. Triggers STT when a segment completes."""
        if time.monotonic() < self._deafen_until:
            return

        # Wake word streaming detection (parallel to VAD)
        if self._wake_detector and self._wake_detector.enabled:
            prev_state = self._wake_state
            self._wake_detector.feed(chunk)
            # Notify if wake word was just detected (must be in async context)
            if prev_state != self._wake_state and self._wake_state == WakeState.WOKEN:
                if self._on_wake_notify:
                    await self._on_wake_notify()

        # WOKEN timeout check
        if self._wake_state == WakeState.WOKEN:
            if time.monotonic() - self._woken_at > self.WOKEN_TIMEOUT:
                self._wake_state = WakeState.AWAITING
                logger.debug("WOKEN timeout → AWAITING")

        is_speech = self._is_speech(chunk)

        if is_speech:
            self._silence_start = None
            if not self._speaking:
                self._speaking = True
                self._speech_chunk_count = 0
                self._interrupt_fired = False
            self._speech_chunk_count += 1
            self._chunks.append(chunk)

            # Transition WOKEN → LISTENING on first speech
            if self._wake_state == WakeState.WOKEN:
                self._wake_state = WakeState.LISTENING

            if not self._interrupt_fired and self._speech_chunk_count >= self.MIN_SPEECH_CHUNKS_FOR_INTERRUPT:
                now = time.monotonic()
                if now - self._last_interrupt >= self._interrupt_cooldown:
                    self._last_interrupt = now
                    self._interrupt_fired = True
                    if self._on_interrupt:
                        await self._on_interrupt()
        elif self._speaking:
            now = time.monotonic()
            if self._silence_start is None:
                self._silence_start = now
                self._chunks.append(chunk)
            elif now - self._silence_start >= self._silence_dur:
                await self._process_segment()
```

    async def _process_segment(self) -> None:
        audio = b"".join(self._chunks)
        self._chunks.clear()
        self._speaking = False
        self._speech_chunk_count = 0
        self._interrupt_fired = False
        self._silence_start = None

        min_bytes = int(self._sample_rate * self.MIN_SEGMENT_SECONDS * 2)
        if len(audio) < min_bytes:
            return

        # Gate: if wake words active, only process in LISTENING state
        if self._wake_state is not None and self._wake_state != WakeState.LISTENING:
            return

        if self._stt_transcribe is None:
            logger.debug("No STT configured, dropping %d bytes of audio", len(audio))
            return

        wav = pcm_to_wav(audio, sample_rate=self._sample_rate)
        try:
            text = await asyncio.wait_for(
                self._stt_transcribe(wav),
                timeout=30.0,
            )
            if text and text.strip() and self._on_transcription:
                stripped = text.strip()
                await self._on_transcription(stripped)
        except asyncio.TimeoutError:
            logger.warning("STT timed out (segment length=%d bytes)", len(audio))
        except Exception:
            logger.warning("STT transcription failed", exc_info=True)
        finally:
            # Back to AWAITING after STT (pass or fail)
            if self._wake_state == WakeState.LISTENING:
                self._wake_state = WakeState.AWAITING
```
